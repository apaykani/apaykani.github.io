<section class="pg-section" aria-labelledby="research-themes">
  <h2 id="research-themes">Research themes</h2>
  <div class="pg-themes">
    <div class="pg-theme">
      <h3><a href="{{ '/research/#reactive-flows' | relative_url }}">Reactive flows and combustion</a></h3>
      <p>Models of turbulence, chemistry and radiation, including plasma-assisted combustion.</p>
    </div>
    <div class="pg-theme">
      <h3><a href="{{ '/research/#scientific-ml' | relative_url }}">Scientific machine learning</a></h3>
      <p>Efficient models for stiff chemical kinetics, with physical constraints and uncertainty estimates.</p>
    </div>
    <div class="pg-theme">
      <h3><a href="{{ '/research/#sustainable-fuels' | relative_url }}">Hydrogen and ammonia</a></h3>
      <p>Combustion chemistry and spray behaviour for engines, turbines and aviation.</p>
    </div>
    <div class="pg-theme">
      <h3><a href="{{ '/research/#thermal-management' | relative_url }}">Thermal management</a></h3>
      <p>Cooling of electric machines through CFD, experiments and design optimisation.</p>
    </div>
  </div>
</section>
